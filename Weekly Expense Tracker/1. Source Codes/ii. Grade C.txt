
// ASSESSMENT 1 GRADE C 
#include <stdio.h>

int main()
{
    //I Declared all variables
    int food;
    int transport;
    int entertainment;

    int total;
    float average;
    int budget = 40000;

    printf("\n==== SOL ASSESSMENT 1 ====\n");
    printf("===== WEEKLY EXPENSE TRACKER =====\n\n");

    printf("=== GRADE C ====\n");
   
    printf("Enter Food Expenses: ");
    scanf("%d", &food);

    printf("Enter Transport Expenses: ");
    scanf("%d", &transport);

    printf("Enter Entertainment Expenses: ");
    scanf("%d", &entertainment);

    // Validation
    if (food < 0 || transport < 0 || entertainment < 0)
    {

//when users enters the wrong values

        printf("\nInvalid Input! Spending cannot be negative.\n");
        printf("\n  Try again! \n");
        return 0;
     }

//When user enters the right values

     total = food + transport + entertainment;
     average = total/7.0;
     
        printf("\n== RESULT OUTPUT  ====\n");
    printf("FOOD: £%d\n", food);
    printf("TRANSPORT: £%d\n", transport);
    printf("ENTERTAINMENT: £%d\n", entertainment);

    printf("\nTotal Weekly Spending: £%d\n", total);
    printf("Daily Average: £%.2f\n", average);
      
 
// Budget comparison
    if (total <= budget)
    {
        printf(" STATUS: Within Budget\n");
    }
    else
    {
        printf("STATUS: Over Budget\n");
    }
        return 0;
 }
