
#include <stdio.h>
#include <string.h>

#define MAX_ITEMS 100

struct Item
{
    int id;
    char assetTag[30];
    char status;
};

int searchItemById(int id)
{
    FILE *file;
    struct Item item;
    int position = 0;

    file = fopen("loans.txt", "r");

    if(file == NULL)
    {
        return -1;
    }

    while(fscanf(file, "%d %29s %c",
                 &item.id,
                 item.assetTag,
                 &item.status) == 3)
    {
        if(item.id == id)
        {
            fclose(file);
            return position;
        }

        position++;
    }

    fclose(file);

    return -1;
}

int addItem(int id, char assetTag[], char status[])
{
    FILE *file;

    if(searchItemById(id) != -1)
    {
        return 0;
    }

    file = fopen("loans.txt", "a");

    if(file == NULL)
    {
        return 0;
    }

    fprintf(file, "%d %s %c\n",
            id,
            assetTag,
            status[0]);

    fclose(file);

    return 1;
}

}

int updateItem(int id,
               const char newAssetTag[],
               const char newStatus[])
{
    FILE *file;
    struct Item items[MAX_ITEMS];

    int count = 0;
    int i;
    int found = 0;

    
     //  Read all records from the file
     //  into the array.
    

    file = fopen("loans.txt", "r");

    if(file == NULL)
    {
        return 0;
    }

    while(count < MAX_ITEMS &&
          fscanf(file, "%d %29s %c",
                 &items[count].id,
                 items[count].assetTag,
                 &items[count].status) == 3)
    {
        count++;
    }

    fclose(file);

    
      // Search the array for the requested ID.
    

    for(i = 0; i < count; i++)
    {
        if(items[i].id == id)
        {
            strcpy(items[i].assetTag, newAssetTag);
            items[i].status = newStatus[0];

            found = 1;
            break;
        }
    }

    if(found == 0)
    {
        return 0;
    }

    
     //  Rewrite the entire file with
     //  the updated records.
    

    file = fopen("loans.txt", "w");

    if(file == NULL)
    {
        return 0;
    }

    for(i = 0; i < count; i++)
    {
        fprintf(file, "%d %s %c\n",
                items[i].id,
                items[i].assetTag,
                items[i].status);
    }

    fclose(file);

    return 1;
}
